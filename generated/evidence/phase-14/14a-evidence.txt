    # R6 host safety: 0 swap on this box; unbounded parallelism can hard-freeze it.
    BB_NUMBER_THREADS = "10"

    # sstate: MIRROR, not a shared SSTATE_DIR.
    #
    # Diffuse needs the cross toolchain and, more to the point, cuda-toolkit-bin 13.4.1 -
    # both already built and proven in Ravencalc's cache. Reads come from there; writes go
    # to this runtime's own cache, so Ravencalc's is never modified and R10 (sstate is a
    # protected asset) holds by construction. Symoneural-LLM/API/CLI point SSTATE_DIR
    # straight at Ravencalc's cache; Common and this runtime use the mirror form, which is
    # the safer of the two conventions in the estate and costs nothing.
    #
    # Anchor the grep when checking this is set: the stock template ships a COMMENTED
    # "#SSTATE_MIRRORS" line, and an unanchored match reports a mirror that is not active.
    SSTATE_DIR = "/home/google/SymonSaysLLC/Symoneural-Diffuse/sstate-cache-master"
    SSTATE_MIRRORS = "file://.* file:///home/google/SymonSaysLLC/Symoneural-Ravencalc/sstate-cache-master/PATH"
    
    # The estate owns Cython. Without this, python3-pyyaml-native pulls OE-Core's
    # python3-cython-native while a symoneural-* recipe pulls ours, and both install
    # /usr/bin/cygdb3 into the same native sysroot - a hard abort at
    # do_prepare_recipe_sysroot. sd-cli itself is pure C++ and does not reach Python, but
    # Phase 14b brings onnxruntime and rembg into this same build directory, and a
    # local.conf that differs between runtimes is its own hazard (R6/R9). Belongs in the
    # distro conf; symoneural.conf stays inert until a local.conf sets DISTRO (item X2).
    PREFERRED_PROVIDER_python3-cython = "symoneural-cython"
    PREFERRED_PROVIDER_python3-cython-native = "symoneural-cython-native"
